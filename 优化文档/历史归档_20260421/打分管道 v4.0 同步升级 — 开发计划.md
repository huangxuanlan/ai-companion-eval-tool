```

MAX_HISTORY_CHARS = 8000  # 约 5000 tokens

def get_history_context(conv_id: str, current_turn: int, max_turns: int = 10) -> str:
    """从 turn_results 读取 current_turn 之前最近 max_turns 轮的 user_input + ai_output。"""
    conn = get_connection()
    rows = conn.execute(
        """SELECT turn, user_input, ai_output FROM turn_results
           WHERE conversation_id=? AND turn < ?
           ORDER BY turn DESC LIMIT ?""",
        (conv_id, current_turn, max_turns),
    ).fetchall()
    conn.close()
    if not rows:
        return ""
    lines = []
    for r in reversed(rows):  # 从旧到新
        lines.append(f"[用户] {r['user_input']}")
        lines.append(f"[AI] {r['ai_output']}")
    result = "\n".join(lines)
    if len(result) > MAX_HISTORY_CHARS:
        result = result[-MAX_HISTORY_CHARS:]
    return result
```

```python
def _build_history_context(results: list, current_index: int, max_turns: int = 10) -> str:
    """从 results 列表（已有的前序轮次）构建 history_context。"""
    start = max(0, current_index - max_turns)
    history_turns = results[start:current_index]
    if not history_turns:
        return ""
    lines = []
    for r in history_turns:
        lines.append(f"[用户] {r.get('user_input', '')}")
        lines.append(f"[AI] {r.get('ai_output', '')}")
    result = "\n".join(lines)
    if len(result) > 8000:
        result = result[-8000:]
    return result
```

> **设计决策**：`score_conversation` 已经有完整的 `results` 列表，直接从内存构建，不需要查数据库。`get_history_context()` 函数为独立打分脚本和外部调用保留。
